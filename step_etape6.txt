export function formatXaf(amount: number, currency = 'FCFA'): string {
  const formatted = new Intl.NumberFormat('fr-CM', {
    style: 'decimal',
    maximumFractionDigits: 0,
  })
    .format(Math.round(amount))
    .replace(/\s/g, '\u202F');
  return `${formatted}\u202F${currency}`;
}

export function formatDateShort(d: Date): string {
  return new Intl.DateTimeFormat('fr-CM', {
    timeZone: 'Africa/Douala',
    day: '2-digit',
    month: '2-digit',
    year: 'numeric',
  }).format(d);
}

'use client';

import type { PdfDocument, PdfFormat, PdfRenderOptions } from './types';

/**
 * Génère le PDF côté client. Charge @react-pdf/renderer à la demande.
 */
export async function renderPdfToBlob(
  doc: PdfDocument,
  format: PdfFormat,
  options: PdfRenderOptions
): Promise<Blob> {
  const { pdf } = await import('@react-pdf/renderer');

  let element;
  if (format === 'ticket') {
    const { TicketDocument } = await import('./templates/TicketTemplate');
    element = TicketDocument({ doc, options });
  } else {
    const { A4Document } = await import('./templates/A4Template');
    element = A4Document({ doc, options });
  }

  return pdf(element as any).toBlob();
}

export function downloadBlob(blob: Blob, filename: string): void {
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = filename;
  document.body.appendChild(a);
  a.click();
  document.body.removeChild(a);
  URL.revokeObjectURL(url);
}
6.5.1 — lib/pdf/format-preference.ts (client)

'use client';

import type { DocType } from '@prisma/client';
import type { PdfFormat } from './types';

const STORAGE_KEY = 'cashbill:pdf-format-preference:v1';

export function getDefaultFormat(docType: DocType): PdfFormat {
  return docType === 'TICKET' || docType === 'RECU' ? 'ticket' : 'a4';
}

export function getPreferredFormat(docType: DocType): PdfFormat {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    if (raw === 'ticket' || raw === 'a4') return raw;
  } catch {}
  return getDefaultFormat(docType);
}

export function savePreferredFormat(format: PdfFormat): void {
  try {
    localStorage.setItem(STORAGE_KEY, format);
  } catch {}
}

ts
import { z } from 'zod';

const lineSchema = z.object({
  description: z.string().min(1).max(200),
  quantity: z.number().positive().max(10000),
  unitPrice: z.number().nonnegative().max(1_000_000_000),
  photoUrl: z.string().url().optional().nullable(),
});

export const documentPayloadSchema = z.object({
  lines: z.array(lineSchema).min(1).max(200),
  vatRate: z.number().min(0).max(30).nullable().default(null),
  depositPercentage: z.number().min(0).max(100).nullable().default(null),
  deliveryFee: z.number().nonnegative().nullable().default(null),
  notes: z.string().max(1000).nullable().default(null),
});

export type DocumentPayloadInput = z.infer<typeof documentPayloadSchema>;

export type ComputedPayload = {
  lines: Array<{ description: string; quantity: number; unitPrice: number; total: number; photoUrl?: string | null }>;
  subtotal: number;
  vatRate: number | null;
  vatAmount: number | null;
  depositAmount: number | null;
  deliveryFee: number | null;
  notes: string | null;
  total: number;
};

/**
 * Calcule côté serveur. Ne jamais faire confiance au client pour les totaux.
 */
export function computePayload(input: DocumentPayloadInput): ComputedPayload {
  const lines = input.lines.map((l) => ({
    description: l.description,
    quantity: l.quantity,
    unitPrice: l.unitPrice,
    total: Math.round(l.quantity * l.unitPrice),
    photoUrl: l.photoUrl ?? null,
  }));

  const subtotal = lines.reduce((acc, l) => acc + l.total, 0);
  const vatAmount = input.vatRate !== null ? Math.round(subtotal * (input.vatRate / 100)) : null;
  const depositAmount =
    input.depositPercentage !== null
      ? Math.round((subtotal + (vatAmount ?? 0)) * (input.depositPercentage / 100))
      : null;
  const deliveryFee = input.deliveryFee ?? null;
  const total =
    subtotal +
    (vatAmount ?? 0) +
    (deliveryFee ?? 0) -
    (depositAmount ?? 0);

  return {
    lines,
    subtotal,
    vatRate: input.vatRate,
    vatAmount,
    depositAmount,
    deliveryFee,
    notes: input.notes,
    total,
  };
}

import type { NextRequest } from 'next/server';
import { readFile } from 'node:fs/promises';
import path from 'node:path';
import { jsonError } from '@/lib/api-helpers';
import { requireAuth } from '@/lib/api/server';

const UPLOADS_DIR = path.join(process.cwd(), 'uploads');

export async function GET(
  req: NextRequest,
  { params }: { params: Promise<{ filename: string }> }
): Promise<Response> {
  try {
    return await requireAuth(req, async (_req, { user }) => {
      const { filename } = await params;

      // Vérifie que le fichier appartient à l'utilisateur
      if (!filename.startsWith(`${user.id}-`)) {
        return jsonError('FORBIDDEN', 'Accès refusé', 403);
      }

      // Protection contre directory traversal
      if (filename.includes('..') || filename.includes('/')) {
        return jsonError('FORBIDDEN', 'Nom de fichier invalide', 403);
      }

      try {
        const filePath = path.join(UPLOADS_DIR, filename);
        const buffer = await readFile(filePath);
        const ext = path.extname(filename).toLowerCase();
        const contentType = ext === '.png' ? 'image/png' : 'image/jpeg';
        return new Response(new Uint8Array(buffer), {
          headers: {
            'Content-Type': contentType,
            'Cache-Control': 'private, max-age=3600',
          },
        });
      } catch {
        return jsonError('NOT_FOUND', 'Fichier introuvable', 404);
      }
    });
  } catch (err) {
    return jsonError('INTERNAL', 'Erreur interne', 500);
  }
}

import { randomUUID } from 'node:crypto';
import { prisma } from '@/lib/db';
import type { DocType, Locale, Mode, User } from '@prisma/client';
import { generateDocumentNumber } from './numbering';
import { computeContentHash } from './hash';
import { computePayload, type DocumentPayloadInput } from './payload';
import { evaluateDocument, detectRepeatedRoundAmounts } from '@/lib/fraud/rules';
import { generatePublicToken } from './verification';
import { generateQrCode } from './integrity';
import { env } from '@/lib/env';

export type CreateDocumentInput = {
  user: User;
  clientId: string | null;
  type: DocType;
  mode: Mode;
  locale: Locale;
  payload: DocumentPayloadInput;
};

export type CreateDocumentResult =
  | { ok: true; documentId: string; number: string; publicToken: string; qrCodeDataUrl: string }
  | { ok: false; reason: 'BLOCKED'; blockedReasons: string[]; flags: Array<{ reason: string; severity: string }> };

export async function createDocument(input: CreateDocumentInput): Promise<CreateDocumentResult> {
  const computed = computePayload(input.payload);
  const year = new Date().getFullYear();

  // Vérifier fraude AVANT la création
  const fraudEval = await evaluateDocument({
    user: input.user,
    docType: input.type,
    mode: input.mode,
    payload: computed,
    totalAmountFcfa: computed.total,
    clientId: input.clientId,
    contentHash: '', // placeholder, recalculé après numérotation
  });

  if (fraudEval.blocked) {
    return {
      ok: false,
      reason: 'BLOCKED',
      blockedReasons: fraudEval.flags.filter((f) => f.severity === 'CRITICAL').map((f) => f.reason),
      flags: fraudEval.flags.map((f) => ({ reason: f.reason, severity: f.severity })),
    };
  }

  // Transaction : numérotation + hash + insert
  const result = await prisma.$transaction(async (tx) => {
    const number = await generateDocumentNumber(tx, input.user.id, input.type, year);

    const contentHash = computeContentHash({
      userId: input.user.id,
      docType: input.type,
      number,
      locale: input.locale,
      payload: computed,
      createdAt: new Date().toISOString(),
    });

    const doc = await tx.document.create({
      data: {
        userId: input.user.id,
        clientId: input.clientId,
        type: input.type,
        mode: input.mode,
        number,
        locale: input.locale,
        payload: computed,
        contentHash,
        watermarked: !input.user.proUntil || input.user.proUntil <= new Date(),
      },
    });

    const publicToken = generatePublicToken();
    const qrCodeDataUrl = await generateQrCode(publicToken, env.APP_URL);

    await tx.documentVerification.create({
      data: { documentId: doc.id, publicToken, isAuthentic: true },
    });

    await tx.document.update({
      where: { id: doc.id },
      data: { qrCodeUrl: qrCodeDataUrl },
    });

    return { doc, number, contentHash, publicToken, qrCodeDataUrl };
  });

  // Écrire les flags non bloquants
  const repeatedRound = await detectRepeatedRoundAmounts(input.user.id);
  const allFlags = [
    ...fraudEval.flags.filter((f) => f.severity !== 'CRITICAL'),
    ...(repeatedRound ? [repeatedRound] : []),
  ];

  if (allFlags.length > 0) {
    await prisma.fraudFlag.createMany({
      data: allFlags.map((f) => ({
        userId: input.user.id,
        documentId: result.doc.id,
        reason: f.reason,
        severity: f.severity,
        details: f.details ?? undefined,
        autoDetected: true,
      })),
    });

    const hasHigh = allFlags.some((f) => f.severity === 'HIGH');
    if (hasHigh) {
      await prisma.user.update({
        where: { id: input.user.id },
        data: { flaggedForReview: true },
      });
    }
  }

  return {
    ok: true,
    documentId: result.doc.id,
    number: result.number,
    publicToken: result.publicToken,
    qrCodeDataUrl: result.qrCodeDataUrl,
  };
}

import { prisma } from '@/lib/db';
import { REFERRAL_REWARD_DAYS, REFERRAL_MAX_ACCUMULATED_DAYS } from '@/lib/constants';

/**
 * Active le parrainage du user (s'il est filleul) si les conditions sont remplies.
 * Appelé après le 1er document envoyé via WhatsApp.
 */
export async function tryActivateReferral(filleulId: string): Promise<void> {
  const referral = await prisma.referral.findUnique({ where: { filleulId } });
  if (!referral || referral.status !== 'PENDING') return;

  const now = new Date();
  const windowEnd = new Date(referral.createdAt.getTime() + 7 * 24 * 60 * 60 * 1000);
  if (now > windowEnd) {
    await prisma.referral.update({
      where: { id: referral.id },
      data: { status: 'REJECTED' },
    });
    return;
  }

  await prisma.$transaction(async (tx) => {
    await tx.referral.update({
      where: { id: referral.id },
      data: { status: 'REWARDED', parrainRewardedAt: now, triggeredAt: now },
    });

    const parrain = await tx.user.findUnique({ where: { id: referral.parrainId } });
    if (!parrain) return;

    const currentPro = parrain.proUntil && parrain.proUntil > now ? parrain.proUntil : now;
    const newPro = new Date(currentPro.getTime() + REFERRAL_REWARD_DAYS * 24 * 60 * 60 * 1000);
    const maxPro = new Date(now.getTime() + REFERRAL_MAX_ACCUMULATED_DAYS * 24 * 60 * 60 * 1000);

    if (newPro > maxPro) {
      const pending = Math.ceil((newPro.getTime() - maxPro.getTime()) / (24 * 60 * 60 * 1000));
      await tx.user.update({
        where: { id: parrain.id },
        data: { proUntil: maxPro, pendingProDays: parrain.pendingProDays + pending },
      });
    } else {
      await tx.user.update({
        where: { id: parrain.id },
        data: { proUntil: newPro },
      });
    }

    await tx.subscription.create({
      data: {
        userId: parrain.id,
        source: 'REFERRAL',
        startsAt: now,
        endsAt: newPro,
      },
    });
  });
}
